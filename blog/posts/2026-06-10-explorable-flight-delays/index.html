vg = {
  const vg = await import("https://cdn.jsdelivr.net/npm/@uwdata/vgplot@0.26.0/+esm");
  vg.coordinator().databaseConnector(await vg.wasmConnector());
  return vg;
}

Every chart below is backed by a DuckDB instance running in your browser via WebAssembly, querying a 200k-row parquet file. Drag horizontally on any histogram to cross-filter the other two. No server is involved.

brush = {
  const dataUrl = new URL("data.parquet", document.baseURI).href;
  await vg.coordinator().exec(`
    CREATE TABLE IF NOT EXISTS flights AS
    SELECT * FROM '${dataUrl}'
  `);
  return vg.Selection.crossfilter();
}
makeHistogram = (column, label) =>
  vg.plot(
    vg.rectY(vg.from("flights", { filterBy: brush }), {
      x: vg.bin(column),
      y: vg.count(),
      fill: "var(--bs-primary, steelblue)",
      inset: 0.5,
    }),
    vg.intervalX({ as: brush }),
    vg.xDomain(vg.Fixed),
    vg.xLabel(label),
    vg.yLabel("flights"),
    vg.width(680),
    vg.height(180),
    vg.marginLeft(60),
  )
makeHistogram("delay", "arrival delay (min)")
makeHistogram("time", "departure hour")
makeHistogram("distance", "distance (mi)")

How it works: build-time Python wrote data.parquet (see prepare_data.py); the page loads Mosaic vgplot from a CDN and registers the parquet with DuckDB-WASM. The brush is a crossfilter Selection shared by all three plots.
